+/**
+ * 发送认证请求的统一入口。
+ *
+ * 行为：
+ *  1. 探测/读取 `authEncryptionSupported` 标记。
+ *     - `true`  -> 直接走加密路径（POST 加密报文）。
+ *     - `false` -> 直接走明文路径（向后兼容旧后端）。
+ *     - `null`  -> 本次先尝试加密：成功就记住 `true`；失败则回落到明文。
+ *
+ *  2. 加密路径内部已实现：解密失败时清缓存 + 强制重拉公钥 + 重试一次。
+ *     仅在重试仍失败时才把错误往上抛。
+ */
+async function postAuthRequest<T>(
+  endpoint: string,
+  payload: Record<string, unknown>
+): Promise<T> {
+  // 尚未探测或已确认不支持加密 -> 走明文
+  if (authEncryptionSupported === false) {
+    return api.post<T>(endpoint, payload);
+  }
+
+  // 尝试加密路径
+  try {
+    const { key } = await fetchAuthPubkey(getCustomApiHost());
+    const encPayload: EncryptedPayload = encryptAuthPayload(payload, key);
+    const result = await api.post<T>(endpoint, encPayload as unknown as Record<string, unknown>);
+    // 加密成功：标记为支持
+    authEncryptionSupported = true;
+    return result;
+  } catch (err) {
+    // 网络/解析/加密失败 -> 记一次 fallback
+    const reason = err instanceof Error ? err.message : String(err);
+    // 后端解密失败统一返回中性错误信息，这里再尝试一次：
+    // 清缓存 + 强制重拉公钥 + 重新加密。
+    try {
+      const { key } = await fetchAuthPubkey(getCustomApiHost(), true);
+      const encPayload: EncryptedPayload = encryptAuthPayload(payload, key);
+      const result = await api.post<T>(endpoint, encPayload as unknown as Record<string, unknown>);
+      authEncryptionSupported = true;
+      return result;
+    } catch (retryErr) {
+      // 重试仍失败：标记不支持加密，落到明文路径再试一次（兼容老后端）
+      authEncryptionSupported = false;
+      // eslint-disable-next-line no-console
+      console.warn(
+        `[authCrypto] Encrypted auth request failed (${reason}); falling back to plaintext.`
+      );
+      return api.post<T>(endpoint, payload);
+    }
+  }
